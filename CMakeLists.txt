cmake_minimum_required(VERSION 3.16)
project(PhaseBoxSim CXX)
include(cmake/version.cmake)   # PHASEBOX_VERSION, from the latest v* git tag
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.0.tar.gz)
FetchContent_MakeAvailable(raylib)
include_directories(include firmware/include)
add_executable(phase_box_native_sim src/native_sim_main.cpp src/PhaseOscillator.cpp)
target_link_libraries(phase_box_native_sim PRIVATE raylib)
target_compile_definitions(phase_box_native_sim PRIVATE PHASEBOX_VERSION="${PHASEBOX_VERSION}")
if(APPLE)
    target_link_libraries(phase_box_native_sim PRIVATE "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo" "-framework CoreMIDI" "-framework CoreFoundation" "-framework CoreAudio")
endif()
