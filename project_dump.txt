namespace phasebox::ui {

void UIController::init() {
    m_current_screen = Screen::DASHBOARD;
    m_coarse_mode = false;
    m_last_turn_us = 0;
}

void UIController::handle_encoder_turn(int delta, uint32_t current_time_us) {
    if (delta == 0) return;

    if (m_current_screen == Screen::BPM_SETTING) {
        handle_bpm_encoder_input(delta, current_time_us);
    } else if (m_current_screen == Screen::PRESET_MENU) {
        // Preset navigation
        m_selected_preset_slot = std::clamp(m_selected_preset_slot + delta, 1, 8);
    }
}

void UIController::handle_bpm_encoder_input(int delta, uint32_t current_time_us) {
    if (delta == 0) return; // don't let idle polling reset the turn-velocity timer
    uint32_t interval_ms = 100;
    if (m_last_turn_us > 0 && current_time_us > m_last_turn_us) {
        interval_ms = (current_time_us - m_last_turn_us) / 1000;
    }
    m_last_turn_us = current_time_us;

    float current_bpm = m_engine.get_bpm();
    float step = 0.1f;

    // Dynamic velocity acceleration
    if (m_coarse_mode) {
        step = (interval_ms < 30) ? 5.0f : 1.0f;
    } else {
        step = (interval_ms < 30) ? 1.0f : 0.1f;
    }

    float new_bpm = current_bpm + (delta * step);

    // Enforce bounds: 0.1 BPM min to 300.0 BPM max
    new_bpm = std::clamp(new_bpm, 0.1f, 300.0f);

    m_engine.set_bpm(new_bpm);
}

void UIController::handle_encoder_click() {
    if (m_current_screen == Screen::DASHBOARD) {
        m_current_screen = Screen::BPM_SETTING;
    } else if (m_current_screen == Screen::BPM_SETTING) {
        // Toggle between Coarse and Fine adjustment modes on BPM screen
        m_coarse_mode = !m_coarse_mode;
    }
}

void UIController::handle_tap_tempo(uint32_t tap_time_us) {
    if (m_last_tap_us > 0 && tap_time_us > m_last_tap_us) {
        uint32_t interval_us = tap_time_us - m_last_tap_us;
        uint32_t interval_ms = interval_us / 1000;

        // Valid tap tempo window: 200 ms (300 BPM) to 3000 ms (20 BPM)
        if (interval_ms >= 200 && interval_ms <= 3000) {
            float calculated_bpm = 60000.0f / static_cast<float>(interval_ms);
            m_engine.set_bpm(calculated_bpm);
            m_engine.reset_phase(); // Align phase accumulator to tap
        }
    }
    m_last_tap_us = tap_time_us;
}

class TerminalScopeGuard {
    struct termios old_t_;
public:
    TerminalScopeGuard() {
        tcgetattr(STDIN_FILENO, &old_t_);
        struct termios new_t = old_t_;
        new_t.c_lflag &= ~(ICANON | ECHO);
        tcsetattr(STDIN_FILENO, TCSANOW, &new_t);
    }
    ~TerminalScopeGuard() {
        tcsetattr(STDIN_FILENO, TCSANOW, &old_t_);
    }
};

bool kbhit() {
    struct timeval tv = {0, 0};
    fd_set fds;
    FD_ZERO(&fds);
    FD_SET(STDIN_FILENO, &fds);
    return select(STDIN_FILENO + 1, &fds, NULL, NULL, &tv) > 0;
}

inline std::string render_oled_screen(int wave_type, float phase_offset, int bpm, int encoder_val, bool button_state, float adc_val) {
    const int width = 32;
    const int height = 7;
    std::vector<std::string> grid(height, std::string(width, ' '));
    
    for (int x = 0; x < width; ++x) {
        float t = (static_cast<float>(x) / width) * 2.0f * 3.14159265f + phase_offset;
        float val = 0.0f;
        switch (wave_type % 4) {
            case 0: val = std::sin(t); break;
            case 1: val = 1.0f - std::fmod(t / 3.14159265f, 2.0f); break;
            case 2: val = (std::sin(t) >= 0.0f) ? 1.0f : -1.0f; break;
            case 3: val = 2.0f * std::abs(2.0f * (t / (2.0f * 3.14159265f) - std::floor(t / (2.0f * 3.14159265f) + 0.5f))) - 1.0f; break;
        }
        int y = static_cast<int>((1.0f - (val + 1.0f) * 0.5f) * (height - 1));
        if (y >= 0 && y < height) grid[y][x] = '*';
    }

    const char* wave_names[] = {"SINE", "SAW", "SQR", "TRI"};
    
    std::string oled = "+--------------------------------+\n";
    char header_buf[64];
    snprintf(header_buf, sizeof(header_buf), "| W:%-4s B:%3d E:%3d%s |\n", wave_names[wave_type%4], bpm, encoder_val, button_state ? "*" : " ");
    oled += header_buf;
    oled += "+--------------------------------+\n";
    
    for (const auto& row : grid) {
        oled += "|" + row + "|\n";
    }
    
    oled += "+--------------------------------+\n";
    char footer_buf[64];
    snprintf(footer_buf, sizeof(footer_buf), "| ADC:%4.2f                       |\n", adc_val);
    oled += footer_buf;
    oled += "+--------------------------------+\n";
    
    return oled;
}

int main() {
    TerminalScopeGuard term_guard;
    bool running = true;
    int current_wave = 0;
    float phase_anim = 0.0f;
    int encoder_val = 50;
    bool button_state = false;
    float adc_expression = 0.0f;
    int current_bpm = 120;
    std::vector<std::chrono::steady_clock::time_point> tap_times;

    std::cout << "\n\n\n\n\n\n\n\n\n\n\n\n\n\n\n\n\n\n";

    while (running) {
        if (kbhit()) {
            int c = getchar();
            if (c != EOF) {
                if (c == 'w' || c == 'W') {
                    current_wave = (current_wave + 1) % 4;
                } else if (c == 's' || c == 'S') {
                    current_wave = (current_wave + 3) % 4;
                } else if (c == 'e' || c == 'E') {
                    button_state = !button_state;
                } else if (c == 't' || c == 'T') {
                    auto now = std::chrono::steady_clock::now();
                    if (!tap_times.empty() && std::chrono::duration_cast<std::chrono::milliseconds>(now - tap_times.back()).count() > 2500) {
                        tap_times.clear();
                    }
                    tap_times.push_back(now);
                    if (tap_times.size() > 4) tap_times.erase(tap_times.begin());
                    if (tap_times.size() >= 2) {
                        double total_ms = 0;
                        for (size_t i = 1; i < tap_times.size(); ++i) {
                            total_ms += std::chrono::duration_cast<std::chrono::milliseconds>(tap_times[i] - tap_times[i-1]).count();
                        }
                        double avg_ms = total_ms / (tap_times.size() - 1);
                        if (avg_ms > 0) {
                            current_bpm = static_cast<int>(60000.0 / avg_ms);
                            if (current_bpm < 40) current_bpm = 40;
                            if (current_bpm > 300) current_bpm = 300;
                        }
                    }
                } else if (c == '1') {
                    adc_expression = 0.0f;
                } else if (c == '2') {
                    adc_expression = 0.5f;
                } else if (c == '3') {
                    adc_expression = 1.0f;
                } else if (c == 'q' || c == 'Q') {
                    running = false;
                }
            }
        }

        phase_anim += (current_bpm / 120.0f) * 0.15f;
        encoder_val = (current_wave * 25) + (int)(std::sin(phase_anim) * 5.0f);
        
        std::cout << "\033[18A";
        
        std::cout << "=== PHASE BOX EMBEDDED OLED SIMULATOR ===\n";
        std::cout << render_oled_screen(current_wave, phase_anim, current_bpm, encoder_val, button_state, adc_expression);
        std::cout << "Controls: [w/s] Wave [t] Tap [e] Btn [1-3] ADC [q] Quit\n" << std::flush;

        usleep(50000);
    }

    std::cout << "\nExiting simulator.\n";
    return 0;
}

void HalEncoder::init(uint8_t pin_a, uint8_t pin_b, uint8_t pin_sw) {
    pin_a_ = pin_a;
    pin_b_ = pin_b;
    pin_sw_ = pin_sw;

#ifndef HOST_BUILD
    gpio_init(pin_a_);
    gpio_set_dir(pin_a_, GPIO_IN);
    gpio_pull_up(pin_a_);

    gpio_init(pin_b_);
    gpio_set_dir(pin_b_, GPIO_IN);
    gpio_pull_up(pin_b_);

    gpio_init(pin_sw_);
    gpio_set_dir(pin_sw_, GPIO_IN);
    gpio_pull_up(pin_sw_);

    uint8_t a = gpio_get(pin_a_);
    uint8_t b = gpio_get(pin_b_);
    prev_state_ = (a << 1) | b;
#endif
}

void HalEncoder::update(uint32_t current_time_us) {
#ifndef HOST_BUILD
    // Read hardware pins
    uint8_t a = gpio_get(pin_a_);
    uint8_t b = gpio_get(pin_b_);
    uint8_t curr_state = (a << 1) | b;

    // Gray-code quadrature state transition check
    if (curr_state != prev_state_) {
        if ((prev_state_ == 0b00 && curr_state == 0b01) ||
            (prev_state_ == 0b01 && curr_state == 0b11) ||
            (prev_state_ == 0b11 && curr_state == 0b10) ||
            (prev_state_ == 0b10 && curr_state == 0b00)) {
            
            // Turn acceleration: fast turns (< 30ms interval) step by 2
            uint32_t interval = current_time_us - last_turn_time_us_;
            int step = (interval < 30000) ? 2 : 1;
            delta_accum_ += step;
            last_turn_time_us_ = current_time_us;

        } else if ((prev_state_ == 0b00 && curr_state == 0b10) ||
                   (prev_state_ == 0b10 && curr_state == 0b11) ||
                   (prev_state_ == 0b11 && curr_state == 0b01) ||
                   (prev_state_ == 0b01 && curr_state == 0b00)) {
            
            uint32_t interval = current_time_us - last_turn_time_us_;
            int step = (interval < 30000) ? 2 : 1;
            delta_accum_ -= step;
            last_turn_time_us_ = current_time_us;
        }
        prev_state_ = curr_state;
    }

    // Active-low button check (pull-up resistor enabled)
    bool raw_sw = !gpio_get(pin_sw_);
    if (raw_sw && !button_pressed_) {
        button_pressed_ = true;
        press_start_time_us_ = current_time_us;
        button_hold_time_ms_ = 0;
    } else if (raw_sw && button_pressed_) {
        button_hold_time_ms_ = (current_time_us - press_start_time_us_) / 1000;
    } else if (!raw_sw && button_pressed_) {
        button_pressed_ = false;
        button_hold_time_ms_ = 0;
    }
#endif
}

int HalEncoder::get_delta() {
    int ret = delta_accum_;
    delta_accum_ = 0;
    return ret;
}

float HalAdc::read_normalized() {
    uint16_t raw = read_raw();
    
    // Normalize raw 12-bit reading (0-4095) to [0.0f, 1.0f]
    float raw_norm = static_cast<float>(raw) / 4095.0f;
    if (raw_norm < 0.0f) raw_norm = 0.0f;
    if (raw_norm > 1.0f) raw_norm = 1.0f;

    // Apply Exponential Moving Average (EMA) filtering first so step response works
    smoothed_val_ = (alpha_ * raw_norm) + ((1.0f - alpha_) * smoothed_val_);

    // Deadband clamping at extremes after smoothing
    if (smoothed_val_ < 0.01f) return 0.0f;
    if (smoothed_val_ > 0.95f) return 1.0f;

    return smoothed_val_;
}

#include "PhaseOscillator.hpp"
void PhaseOscillator::update(float dt) {
    float freq = (bpm / 60.0f) * 2.0f;
    phase += freq * dt * static_cast<float>(M_PI);
    if (phase >= 2.0f * static_cast<float>(M_PI)) phase -= 2.0f * static_cast<float>(M_PI);
}
float PhaseOscillator::evaluateCC(float t_offset, float adc_expression) {
    float mod_phase = (phase * pm_ratio) + t_offset;
    float modulator = std::sin(mod_phase) * pm_depth;
    float t = phase + t_offset + phase_offset + modulator;
    t = std::fmod(t, 2.0f * static_cast<float>(M_PI));
    if (t < 0.0f) t += 2.0f * static_cast<float>(M_PI);
    float val = shapeValue(t);
    val *= (0.5f + 0.5f * adc_expression);
    return (val * 0.5f) + 0.5f;
}

float PhaseOscillator::shapeValue(float t) const {
    switch (shape) {
        case SINE: return std::sin(t);
        case SAW: return 1.0f - (t / static_cast<float>(M_PI));
        case SQUARE: return (t < static_cast<float>(M_PI)) ? 1.0f : -1.0f;
        case TRIANGLE: {
            float norm = t / (2.0f * static_cast<float>(M_PI));
            return 2.0f * std::abs(2.0f * (norm - std::floor(norm + 0.5f))) - 1.0f;
        }
    }
    return 0.0f;
}

float PhaseOscillator::normalizedAt(float p) const {
    const float two_pi = 2.0f * static_cast<float>(M_PI);
    float modulator = std::sin(p * pm_ratio) * pm_depth;
    float t = std::fmod(p + phase_offset + modulator, two_pi);
    if (t < 0.0f) t += two_pi;
    return (shapeValue(t) * 0.5f) + 0.5f;
}

namespace phasebox::storage {

bool PresetManager::init() {
    // Hardware LittleFS initialization happens here on Pico build
    return true;
}

bool PresetManager::save_preset(uint8_t slot, const PresetPatch& patch) {
    if (!is_valid_slot(slot)) return false;

    char filename[64];
    std::snprintf(filename, sizeof(filename), "patch%d.bin", slot);

    FILE* file = std::fopen(filename, "wb");
    if (!file) return false;

    size_t written = std::fwrite(&patch, sizeof(PresetPatch), 1, file);
    std::fclose(file);

    return written == 1;
}

bool PresetManager::load_preset(uint8_t slot, PresetPatch& patch) {
    if (!is_valid_slot(slot)) return false;

    char filename[64];
    std::snprintf(filename, sizeof(filename), "patch%d.bin", slot);

    FILE* file = std::fopen(filename, "rb");
    if (!file) return false;

    size_t read_bytes = std::fread(&patch, sizeof(PresetPatch), 1, file);
    std::fclose(file);

    // Validate version tag
    if (read_bytes != 1 || patch.version != 1) {
        return false;
    }

    return true;
}

} // namespace phasebox::storage

// The 128x64 OLED shows exactly what the active destination emits: waveform shape (P1), phase
// offset (P3), phase modulation (P6), clock multiplier (P10), Turing blend (P11), weight (P9).
void DrawOLEDDisplay(RenderTexture2D& oled_target, const PhaseBoxState& state, int active_dest,
                     const PresetSession& session, uint32_t now_ms) {
    BeginTextureMode(oled_target);
    ClearBackground(BLACK);

    DrawRectangleLines(0, 0, 128, 64, WHITE);
    DrawLine(0, 14, 128, 14, WHITE);
    DrawLine(0, 50, 128, 50, WHITE);

    std::string header = simui::page_name(state.active_page);
    header += " D" + std::to_string(active_dest + 1);
    DrawText(header.c_str(), 4, 3, 10, WHITE);
    // Preset slot, with '*' while there are edits that are not saved yet.
    std::string slot_tag = "S" + std::to_string(session.slot()) + (session.dirty() ? "*" : "");
    DrawText(slot_tag.c_str(), 104, 3, 10, WHITE);

    const int wave_width = 120;
    const int wave_height = 33;
    const int start_x = 4;
    const int start_y = 16;

    std::vector<float> curve = simui::preview_curve(state, active_dest, wave_width);
    std::vector<Vector2> points;
    points.reserve(curve.size());
    for (int x = 0; x < wave_width; ++x) {
        float py = start_y + wave_height - (curve[x] * wave_height);
        points.push_back({(float)(start_x + x), py});
    }
    for (size_t i = 1; i < points.size(); ++i) {
        DrawLineV(points[i-1], points[i], WHITE);
    }
    for (int mx : simui::preview_markers(state, active_dest, wave_width)) {
        mx = std::clamp(mx, 0, wave_width - 1);
        DrawCircle(start_x + mx, (int)points[mx].y, 2.0f, WHITE);
    }

    // Hold progress grows along the divider; a save/load message replaces the footer for 1.5 s.
    float hold = session.hold_progress(now_ms);
    if (hold > 0.0f) DrawRectangle(0, 49, (int)(128 * hold), 3, WHITE);
    const char* status = session.status(now_ms);
    std::string footer = status ? std::string(status) : simui::footer(state, state.active_page, active_dest);
    DrawText(footer.c_str(), 4, 53, 8, WHITE);

    EndTextureMode();
}

int main(int argc, char** argv) {
    // `phase_box_native_sim --list-midi` prints what CoreMIDI reports (works even if Audio MIDI Setup is blank).
    for (int i = 1; i < argc; ++i) {
        if (std::string(argv[i]) == "--list-midi") {
#ifdef __APPLE__
            CoreMidiBackend backend;
            backend.print_all(stdout);
#else
            std::printf("MIDI output is macOS only.\n");
#endif
            return 0;
        }
    }

    const int screenWidth = 800;
    const int screenHeight = 680;
    InitWindow(screenWidth, screenHeight, "Phase Box - Polyrhythmic Macro Simulator");
    SetTargetFPS(60);

    PhaseBoxState state;
    state.turing.seed(static_cast<uint32_t>(std::chrono::steady_clock::now().time_since_epoch().count()));
    int active_dest = 0;
    TapTempo tap_tempo;

    // Presets are files in $PHASE_BOX_PRESET_DIR (default: the current directory): preset_1.bin .. preset_8.bin
    static std::string preset_dir;
    const char* env_dir = std::getenv("PHASE_BOX_PRESET_DIR");
    preset_dir = (env_dir && *env_dir) ? env_dir : "./";
    if (preset_dir.back() != '/') preset_dir += '/';
    PresetManager::path_prefix = preset_dir.c_str();
    FilePresetStorage preset_storage;
    PresetSession session(preset_storage);
    double clock_ms = 0.0;                   // frame-time clock: deterministic and wrap-free
    session.load(state, 0);                  // settings restore on power-up

#ifdef __APPLE__
    // MIDI goes straight to the hardware destination named by $PHASE_BOX_MIDI_DEST (default "Melodee FM-1").
    // No virtual port is created and no MIDI Studio patching is needed. PHASE_BOX_MIDI_DEST=virtual brings
    // back the old "Phase Box Virtual Out" source.
    CoreMidiBackend midi_backend;
    midi_route::Router midi(midi_backend, midi_route::Config::from_env(std::getenv("PHASE_BOX_MIDI_DEST")));
    MidiChangeFilter midi_filter;           // like the device: send on change, plus a 1 s refresh
    uint32_t midi_epoch = midi.epoch();
    if (!midi_backend.ok()) std::printf("[midi] CoreMIDI client could not be created (status %d)\n", midi_backend.error());
#endif

    RenderTexture2D oled_target = LoadRenderTexture(128, 64);
    uint32_t last_cc_ms = 0;                 // MIDI goes out every 30 ms of frame time

    while (!WindowShouldClose()) {
        float dt = GetFrameTime();
        clock_ms += dt * 1000.0;
        const uint32_t now_ms = static_cast<uint32_t>(clock_ms);

        if (IsKeyPressed(KEY_E)) state.active_page = simui::next_page(state.active_page, +1);
        if (IsKeyPressed(KEY_Q)) state.active_page = simui::next_page(state.active_page, -1);

        if (IsKeyPressed(KEY_W) || IsKeyPressed(KEY_UP))   simui::adjust(state, active_dest, state.active_page, +1);
        if (IsKeyPressed(KEY_S) || IsKeyPressed(KEY_DOWN)) simui::adjust(state, active_dest, state.active_page, -1);

        if (IsKeyPressed(KEY_M)) state.step_turing(); // same as the Mutate footswitch
        if (IsKeyPressed(KEY_R)) session.reset_to_defaults(state, now_ms);   // everything back to defaults
        if (IsKeyPressed(KEY_U)) session.undo_reset(state, now_ms);          // ...and back again (until the next edit)
        if (IsKeyPressed(KEY_LEFT_BRACKET))  session.select_slot(session.slot() - 1, state, now_ms);
        if (IsKeyPressed(KEY_RIGHT_BRACKET)) session.select_slot(session.slot() + 1, state, now_ms);

        if (IsKeyPressed(KEY_T) || IsKeyPressed(KEY_SPACE)) {
            float tapped_bpm;
            if (tap_tempo.tap(now_ms, tapped_bpm)) state.osc.bpm = tapped_bpm;
            state.restart_phase();   // every tap restarts the LFOs at 0 degrees (Footswitch 1)
        }

        if (IsKeyPressed(KEY_ONE)) state.adc_expression = 0.0f;
        if (IsKeyPressed(KEY_TWO)) state.adc_expression = 0.5f;
        if (IsKeyPressed(KEY_THREE)) state.adc_expression = 1.0f;

        state.advance(dt); // master phase; Turing steps on each master cycle

#ifdef __APPLE__

// UART Configuration for Standard MIDI (31,250 baud)
#define MIDI_UART_ID   uart0
#define MIDI_BAUD_RATE 31250
#define MIDI_TX_PIN    0
#define MIDI_RX_PIN    1

using namespace phasebox::core;
using namespace phasebox::hal;
using namespace phasebox::storage;
using namespace phasebox::ui;

// Global Instances for Dual-Core Operations
static ModulationEngine g_engine;
static MacroEngine      g_macro;
static HalAdc           g_adc;

static HalEncoder       g_encoder;
static HalDisplay       g_display;
static phasebox::ui::UIController       g_ui_controller(g_engine);

// Forward declaration for Core 1 UI loop
void core1_main();

/**
 * @brief Core 0 Hardware Timer Callback (1000 Hz / 1ms period)
 * Driven by the RP2040 hardware timer alarm.
 */
bool repeating_timer_callback(struct repeating_timer *t) {
    uint32_t current_time_us = time_us_32();

    // 1. Read smoothed expression pedal ADC voltage [0.0f - 1.0f]
    float expr_val = g_adc.read_normalized();
    g_macro.set_expression_value(expr_val);

    // 2. Tick engine (updates LFO phase accumulators, tap clock, and CC mapping)
    auto msgs = g_engine.tick(current_time_us);

    // 3. Apply macro scaling matrix & transmit MIDI CC over UART0
    for (size_t i = 0; i < msgs.size(); ++i) {
        uint8_t scaled_data2 = g_macro.apply(i, msgs[i].data2);

        uart_putc_raw(MIDI_UART_ID, msgs[i].status);
        uart_putc_raw(MIDI_UART_ID, msgs[i].data1);
        uart_putc_raw(MIDI_UART_ID, scaled_data2);
    }

    return true; // Keep timer repeating
}

int main() {
    // 1. Initialize Pico stdio and HAL drivers
    stdio_init_all();

    // 2. Configure UART0 for 31,250 baud MIDI
    uart_init(MIDI_UART_ID, MIDI_BAUD_RATE);
    gpio_set_function(MIDI_TX_PIN, GPIO_FUNC_UART);
    gpio_set_function(MIDI_RX_PIN, GPIO_FUNC_UART);

    // 3. Initialize ADC Hardware
    g_adc.init();

    // 4. Launch Core 1 for OLED display, rotary encoder, and flash storage
    multicore_launch_core1(core1_main);

    // 5. Set initial engine parameters
    g_engine.set_bpm(120.0f);

    // 6. Attach 1000 Hz repeating timer interrupt to Core 0 (-1000 us = fixed frequency)
    struct repeating_timer timer;
    add_repeating_timer_us(-1000, repeating_timer_callback, NULL, &timer);

    // 7. Core 0 Idle Loop: Process non-blocking incoming MIDI bytes (e.g. MIDI Clock 0xF8)
    while (true) {
        if (uart_is_readable(MIDI_UART_ID)) {
            uint8_t byte = uart_getc(MIDI_UART_ID);
            if (byte == 0xF8) {
                // Incoming MIDI Clock pulse (24 PPQN sync handling)
            }
        }
        tight_loop_contents();
    }

    return 0;
}

/**
 * @brief Core 1 Entry Point
 * Handles display rendering, rotary menu state transitions, and flash saves.
 */
void core1_main() {
    // Initialize Core 1 peripherals
    g_encoder.init(hal::pins::ENC_A, hal::pins::ENC_B, hal::pins::ENC_SW);
    g_display.init(hal::pins::OLED_SDA, hal::pins::OLED_SCL);
    g_ui_controller.init();

    while (true) {
        uint32_t now_us = time_us_32();

        // 1. Poll rotary encoder hardware
        g_encoder.update(now_us);
        int delta = g_encoder.get_delta();
        bool btn = g_encoder.is_button_pressed();
        uint32_t hold_ms = g_encoder.get_button_hold_time_ms();

        // 2. Update UI state machine
        g_ui_controller.handle_encoder_turn(delta, time_us_32());

        // 3. Render frame to local display framebuffer and send over I2C
        g_display.clear();
        // g_ui_controller.render();
        g_display.render();

        // 4. Limit to ~30 FPS (33 ms frame delay)
        sleep_ms(33);
    }
}

int count_rising_edges(ModulationEngine& e, int ticks) {
    e.get_lfo(0).set_waveform(Waveform::Square);
    int edges = 0;
    int prev = e.tick(1000)[0].data2;
    for (int i = 1; i <= ticks; ++i) {

        if (prev == 0 && v == 127) ++edges;
        prev = v;
    }
    return edges;
}

    int list_calls = 0, virtual_created = 0;
    bool send_fails = false, virtual_ok = true;

    std::vector<Endpoint> destinations() override { ++list_calls; return list; }
    bool send(uint64_t h, const uint8_t* b, size_t n) override {
        if (send_fails) return false;
        // like CoreMIDI: sending to an endpoint that no longer exists fails
        if (std::none_of(list.begin(), list.end(), [&](const Endpoint& e) { return e.handle == h && !e.offline; })) return false;
        sent.push_back({h, std::vector<uint8_t>(b, b + n)});
        return true;
    }
    bool create_virtual_source(const std::string&) override { ++virtual_created; return virtual_ok; }
    bool send_virtual(const uint8_t* b, size_t n) override { virtual_sent.emplace_back(b, b + n); return true; }

    void add(uint64_t h, std::vector<std::string> names, bool offline = false) { list.push_back({h, std::move(names), offline}); }
    void remove(uint64_t h) { list.erase(std::remove_if(list.begin(), list.end(), [&](const Endpoint& e) { return e.handle == h; }), list.end()); }
    size_t sent_to(uint64_t h) const { return std::count_if(sent.begin(), sent.end(), [&](const Sent& s) { return s.handle == h; }); }
};
bool has(const std::string& s, const char* part) { return s.find(part) != std::string::npos; }
} // namespace

TEST(SeqlockBufferTest, ReadersNeverSeeATornFrameWhileAWriterKeepsUpdating) {
    // Like the firmware: one writer publishing frames with idle time in between, other threads reading.
    // (A writer that never pauses would starve any seqlock reader, so this test must not do that.)
    constexpr size_t N = 1024;
    SeqlockBuffer<N> shared;
    std::atomic<bool> stop{false};
    std::thread writer([&] {
        std::vector<uint8_t> frame(N);
        for (uint32_t i = 1; !stop.load(); ++i) {
            std::memset(frame.data(), static_cast<int>(i & 0xFF), N);   // every byte of a frame is the same value
            shared.write(frame.data());
            std::this_thread::sleep_for(std::chrono::microseconds(20));
        }
    });

    std::atomic<long> good{0}, torn{0}, retried{0};
    auto reader = [&] {
        std::vector<uint8_t> out(N);
        const auto until = std::chrono::steady_clock::now() + std::chrono::milliseconds(300);
        long g = 0, t = 0, r = 0;
        while (std::chrono::steady_clock::now() < until) {
            if (!shared.read(out.data())) { ++r; continue; }
            bool uniform = true;
            for (size_t k = 1; k < N; ++k) uniform = uniform && out[k] == out[0];
            uniform ? ++g : ++t;
        }
        good += g; torn += t; retried += r;
    };
    std::thread r1(reader), r2(reader);
    reader();
    r1.join(); r2.join();
    stop = true;
    writer.join();

    EXPECT_EQ(torn.load(), 0L);          // the property that matters: a reader never gets a mixed frame
    EXPECT_GE(good.load(), 100L);        // and with a paced writer readers do get frames
    (void)retried;
}

    EXPECT_TRUE(b.read(out, &s0));

#include <gtest/gtest.h>
#include <cstring>
#include <map>
#include <string>
#include <vector>

namespace {
struct Recorder : MidiOut {
    struct Msg { uint8_t ch, cc, val; uint32_t at_ms; };
    std::vector<Msg> sent;
    uint32_t* clock = nullptr;
    void cc(uint8_t ch, uint8_t c, uint8_t v) override { sent.push_back({ch, c, v, clock ? *clock : 0}); }
};
struct MemStorage : PresetStorage {
    std::map<int, std::vector<uint8_t>> slots; int writes = 0;
    bool write(int slot, const uint8_t* d, size_t n) override { ++writes; slots[slot].assign(d, d + n); return true; }
    size_t read(int slot, uint8_t* out, size_t cap) override {
        auto it = slots.find(slot);
        if (it == slots.end() || it->second.size() > cap) return 0;
        std::memcpy(out, it->second.data(), it->second.size());
        return it->second.size();
    }
};

// A rig that advances time 1 ms per step, polling the controls every step and running the 60 Hz frame on schedule.
struct Rig {
    PhaseBoxState state;
    MemStorage mem;
    NoStorage none;
    PresetSession presets;
    Recorder midi;
    DeviceApp app;
    DeviceInputs in;
    uint64_t now_us = 1000000;      // start at 1 s, as if booted a moment ago
    uint64_t next_frame_us = 1000000;
    uint32_t ms_clock = 1000;

    explicit Rig(bool with_storage = true, uint16_t oled_h = 32)
        : presets(with_storage ? static_cast<PresetStorage&>(mem) : static_cast<PresetStorage&>(none)),
          app(state, presets, midi, 128, oled_h) { midi.clock = &ms_clock; }

    void step() {
        now_us += 1000; ms_clock = static_cast<uint32_t>(now_us / 1000);
        app.poll_inputs(in, now_us);
        if (now_us >= next_frame_us) { app.frame(now_us); next_frame_us += DeviceApp::FRAME_US; }
    }
    void run_ms(int ms) { for (int i = 0; i < ms; ++i) step(); }

    void detent(int dir) {   // one click of the encoder, 1 ms per signal change
        static const bool cw[4][2]  = {{0,1},{0,0},{1,0},{1,1}};
        static const bool ccw[4][2] = {{1,0},{0,0},{0,1},{1,1}};
        for (int i = 0; i < 4; ++i) { in.enc_a = dir > 0 ? cw[i][0] : ccw[i][0]; in.enc_b = dir > 0 ? cw[i][1] : ccw[i][1]; step(); }
    }
    void press(bool DeviceInputs::*pin, int hold_ms) { in.*pin = true; run_ms(hold_ms); in.*pin = false; run_ms(40); }
};
} // namespace

TEST(Device, TurningTheEncoderEditsTheCurrentPage) {
    Rig r;
    r.run_ms(50);
    EXPECT_EQ(r.state.active_page, 0);
    r.detent(+1);
    EXPECT_EQ(r.state.osc.shape, PhaseOscillator::SAW);
    r.detent(-1); r.detent(-1);
    EXPECT_EQ(r.state.osc.shape, PhaseOscillator::TRIANGLE);   // wraps around
}

TEST(Device, ClickGoesToTheNextPageAndWrapsAndSavesNothing) {
    Rig r;
    r.run_ms(50);
    r.press(&DeviceInputs::enc_sw, 100);

    EXPECT_EQ(r.state.active_page, 0);                          // 11 pages, back to the start
    EXPECT_EQ(r.mem.writes, 0);
}

TEST(Device, HoldingForOneSecondSavesAndDoesNotAlsoChangePage) {
    Rig r;
    r.run_ms(50);
    r.detent(+1);                                               // an edit to save
    r.in.enc_sw = true;
    r.run_ms(900);
    EXPECT_EQ(r.mem.writes, 0);

    EXPECT_GE(r.mem.writes, 1);

    ASSERT_TRUE(PresetManager::decode(m.slots[1].data(), m.slots[1].size(), back));
    EXPECT_EQ(back.osc.shape, PhaseOscillator::SAW);
}

TEST(Device, SlowTurnsAreFineAndFastSpinsAreCoarseOnTheBpmPage) {
    Rig r;
    r.run_ms(50);
    r.press(&DeviceInputs::enc_sw, 100);                        // page 2 = BPM
    ASSERT_EQ(r.state.active_page, 1);
    const float start = r.state.osc.bpm;
    for (int i = 0; i < 3; ++i) { r.detent(+1); r.run_ms(150); }
    EXPECT_NEAR(r.state.osc.bpm, start + 3.0f, 0.01f);          // 1 BPM per slow click
    r.run_ms(300);
    const float mid = r.state.osc.bpm;
    for (int i = 0; i < 3; ++i) r.detent(+1);                   // 4 ms apart: a fast spin
    EXPECT_NEAR(r.state.osc.bpm, mid + 1.0f + 5.0f + 5.0f, 0.01f);   // first click is still slow, then 5 per click
    r.run_ms(300);
    r.detent(-1);
    EXPECT_NEAR(r.state.osc.bpm, mid + 11.0f - 1.0f, 0.01f);
}

TEST(Device, TheCurveAndHoldBarAreDrawnOnBothPanelSizes) {
    for (uint16_t h : {uint16_t(32), uint16_t(64)}) {
        Rig r(true, h);
        r.run_ms(100);
        phasebox::hal::FrameBuffer fb(128, h);
        oled::draw(fb, r.state, 0, r.presets, r.ms_clock);
        int lit = 0; for (size_t i = 0; i < fb.size(); ++i) for (int b = 0; b < 8; ++b) lit += (fb.data()[i] >> b) & 1;
        EXPECT_GE(lit, 300);                                    // text, two divider lines and the waveform
        EXPECT_LE(lit, static_cast<int>(fb.size()) * 8 / 2);    // but not a solid block

        r.in.enc_sw = true; r.run_ms(600);                      // hold in progress: bar grows along the top divider
        phasebox::hal::FrameBuffer held(128, h);
        oled::draw(held, r.state, 0, r.presets, r.ms_clock);
        int bar = 0; for (int x = 0; x < 128; ++x) bar += (held.data()[x] >> 2) & 1;   // y = 10 is bit 2 of page 1 -> row below
        (void)bar;
        int on_line = 0; for (int x = 0; x < 128; ++x) on_line += (held.data()[128 + x] >> 2) & 1;   // y=10: page 1, bit 2
        EXPECT_GE(on_line, 60); EXPECT_LE(on_line, 80);          // about 60% of 128 px after 600 of 1000 ms
        r.in.enc_sw = false; r.run_ms(60);
    }
}

TEST(Device, SettingsComeBackAfterAPowerCycle) {
    MemStorage shared;
    {
        PhaseBoxState s; PresetSession p(shared); Recorder m;
        DeviceApp app(s, p, m);
        p.load(s, 0);
        DeviceInputs in; uint64_t t = 1000000;
        auto step = [&](int n) { for (int i = 0; i < n; ++i) { t += 1000; app.poll_inputs(in, t); if (i % 17 == 0) app.frame(t); } };
        step(50);
        static const bool cw[4][2] = {{0,1},{0,0},{1,0},{1,1}};
        for (int i = 0; i < 4; ++i) { in.enc_a = cw[i][0]; in.enc_b = cw[i][1]; step(1); }
        EXPECT_EQ(s.osc.shape, PhaseOscillator::SAW);
        step(2300);                                             // quiet for 2 s: auto-saved
    }
    PhaseBoxState s2; PresetSession p2(shared);
    EXPECT_TRUE(p2.load(s2, 0));
    EXPECT_EQ(s2.osc.shape, PhaseOscillator::SAW);
}

namespace {
struct NullMidi : hal::HalMidi {
    void send_cc(uint8_t, uint8_t, uint8_t) override {}
    bool poll_clock_tick() override { return false; }
};
struct ScriptedGpio : hal::HalGpio {
    bool enc_sw = false;

TEST(Core1, EncoderRoutesToUiAndRenderIsThrottled) {
    core::ModulationEngine engine;
    engine.set_bpm(120.0f);
    ui::UIController ui(engine);
    ScriptedGpio gpio;
    core::Core1Task task(ui, gpio);
    Counter counter;
    task.set_render_callback(count_render, &counter);

    EXPECT_TRUE(ui.get_current_screen() != ui::Screen::DASHBOARD);   // click left the dashboard

    EXPECT_EQ(counter.calls, 2);
}

    std::vector<int> a, b;

TEST(Turing, FullMutationIsDeterministicNotNoise) {
    // Always flipping the feedback bit gives a fixed pattern (period 2x loop length),
    // so 100% is NOT the most random setting.
    TuringMachine t;
    t.mutation_prob = 1.0f;
    for (int i = 0; i < 40; ++i) t.step();
    std::vector<int> a, b;
    for (int i = 0; i < 32; ++i) a.push_back(t.step());
    for (int i = 0; i < 32; ++i) b.push_back(t.step());
    EXPECT_TRUE(a == b);
}

TEST(Turing, HalfMutationVariesAndSeedIsReproducible) {
    TuringMachine x, y;
    x.mutation_prob = y.mutation_prob = 0.5f;
    x.seed(42); y.seed(42);
    std::set<int> seen;
    bool same = true;
    for (int i = 0; i < 256; ++i) { int a = x.step(); int b = y.step(); seen.insert(a); same = same && (a == b); }
    EXPECT_TRUE(same);
    EXPECT_GE(seen.size(), 40u);
    TuringMachine z; z.mutation_prob = 0.5f; z.seed(0);   // seed(0) must not lock the RNG
    std::set<int> seen0; for (int i = 0; i < 256; ++i) seen0.insert(z.step());
    EXPECT_GE(seen0.size(), 40u);
}

TEST(HalAdcTest, EMASmoothingStepResponse) {
    HalAdc adc;
    adc.init();

    // Inject maximum ADC value (4095 -> 1.0f)
    adc.set_simulated_raw(4095);

    // Call 1: alpha = 0.15 -> 0.15 * 1.0 = 0.15
    float val1 = adc.read_normalized();
    EXPECT_NEAR(val1, 0.15f, 0.01f);

    // Call 2: 0.15 * 1.0 + 0.85 * 0.15 = 0.2775
    float val2 = adc.read_normalized();
    EXPECT_NEAR(val2, 0.2775f, 0.01f);

    // Run 30 iterations: smoothed value should converge towards 1.0f
    for (int i = 0; i < 30; ++i) {
        adc.read_normalized();
    }
    EXPECT_FLOAT_EQ(adc.read_normalized(), 1.0f);
}

// Run the shared tick for `steps` steps; return destination `dest`'s MIDI value each step.
std::vector<int> run(PhaseBoxState& s, int dest, int steps, float dt = 0.001f) {
    std::vector<int> v;
    for (int i = 0; i < steps; ++i) {
        s.advance(dt);
        v.push_back(s.evaluateMacros(dt)[dest].value);
    }
    return v;
}
int rising_crossings(const std::vector<int>& v, int threshold) {
    int n = 0;
    for (size_t i = 1; i < v.size(); ++i) if (v[i - 1] <= threshold && v[i] > threshold) ++n;
    return n;
}
PhaseBoxState fresh() { PhaseBoxState s; s.adc_expression = 1.0f; return s; } // 120 BPM: one master cycle = 0.5 s

TEST(SimPages, ClockMultiplierChangesHowManyCyclesTheMidiMakes) {
    auto cycles_over_four_beats = [](float mult) {
        PhaseBoxState s = fresh(); int d = 0;
        s.macros.destinations[0].clock_mult = 1.0f;
        while (s.macros.destinations[0].clock_mult < mult - 1e-3f) adjust(s, d, P_MAC_DIV, +1);
        while (s.macros.destinations[0].clock_mult > mult + 1e-3f) adjust(s, d, P_MAC_DIV, -1);
        return rising_crossings(run(s, 0, 2000), 63);     // 2000 ms = 4 master cycles
    };
    int c1 = cycles_over_four_beats(1.0f), c2 = cycles_over_four_beats(2.0f), ch = cycles_over_four_beats(0.5f);
    EXPECT_GE(c1, 3); EXPECT_LE(c1, 5);       // ~4
    EXPECT_GE(c2, 7); EXPECT_LE(c2, 9);       // ~8
    EXPECT_GE(ch, 1); EXPECT_LE(ch, 3);       // ~2
}

namespace {
std::vector<uint8_t> saved_bytes(const PhaseBoxState& s) {
    uint8_t buf[PresetManager::MAX_BYTES];
    size_t n = PresetManager::encode(s, buf, sizeof(buf));
    return std::vector<uint8_t>(buf, buf + n);
}
void scramble(PhaseBoxState& s) {
    s.osc.shape = PhaseOscillator::SQUARE; s.osc.bpm = 77.0f; s.osc.phase_offset = 2.0f; s.osc.pm_depth = 3.0f;
    s.osc.pm_ratio = 5.0f; s.osc.phase = 4.0f;
    s.turing.mutation_prob = 0.7f; s.turing_mix = 0.6f; s.turing_norm = 0.9f;
    s.midi_channel = 9; s.midi_cc_num = 99;
    s.macros.destinations[0] = {99, 9, -2.0f, 0.9f, 8.0f};
    s.macros.destinations[1].clock_mult = 0.125f; s.macros.destinations[2].midi_channel = 16;
    for (int i = 0; i < 5; ++i) s.step_turing();
    s.advance(0.1f); s.evaluateMacros(0.1f);
}
}

TEST(SimReset, EveryParameterGoesBackToItsDefault) {
    PhaseBoxState s, fresh;
    scramble(s);
    ASSERT_TRUE(saved_bytes(s) != saved_bytes(fresh));       // the scramble really changed things
    s.reset_parameters();
    EXPECT_TRUE(saved_bytes(s) == saved_bytes(fresh));       // identical to a freshly started unit, byte for byte
    EXPECT_NEAR(s.osc.phase, 0.0f, 0.0001f);
    EXPECT_NEAR(s.osc.pm_ratio, 2.0f, 0.0001f);              // the unsaved oscillator settings too
    EXPECT_NEAR(s.turing_norm, 0.5f, 0.0001f);
    EXPECT_NEAR(s.macros.dest_phase(0), 0.0f, 0.0001f);
    EXPECT_EQ(s.macros.destinations.size(), 3u);
    for (int i = 0; i < 8; ++i) EXPECT_EQ(s.turing.step(), fresh.turing.step());   // same Turing pattern as a new unit
}

// Auto-save (2 s after the last edit), hold-to-save (1 s) and slots, with a fake clock and in-memory storage.
#include <gtest/gtest.h>
#include <cstring>
#include <map>
#include <string>
#include <vector>
#include "PresetSession.hpp"

namespace {
struct MemStorage : PresetStorage {
    std::map<int, std::vector<uint8_t>> slots;
    bool fail = false;
    int writes = 0;
    bool write(int slot, const uint8_t* d, size_t n) override {
        ++writes;
        if (fail) return false;
        slots[slot].assign(d, d + n);
        return true;
    }
    size_t read(int slot, uint8_t* out, size_t cap) override {
        auto it = slots.find(slot);
        if (it == slots.end() || it->second.size() > cap) return 0;
        std::memcpy(out, it->second.data(), it->second.size());
        return it->second.size();
    }
};
bool same(const char* a, const char* b) { return a && b && std::strcmp(a, b) == 0; }
float stored_bpm(MemStorage& m, int slot) {
    PhaseBoxState tmp;
    return PresetManager::decode(m.slots[slot].data(), m.slots[slot].size(), tmp) ? tmp.osc.bpm : -1.0f;
}
} // namespace

TEST(AutoSave, NothingIsWrittenWithoutAnEdit) {
    MemStorage m; PresetSession ps(m); PhaseBoxState s;
    for (uint32_t t = 0; t < 30000; t += 16) ps.update(s, t);
    EXPECT_EQ(m.writes, 0);
    EXPECT_FALSE(ps.dirty());
}

TEST(AutoSave, WritesSilentlyAfterTwoQuietSeconds) {
    MemStorage m; PresetSession ps(m); PhaseBoxState s;
    ps.update(s, 1000);
    s.osc.bpm = 140.0f;
    ps.update(s, 1000);
    EXPECT_TRUE(ps.dirty());
    ps.update(s, 2999);
    EXPECT_EQ(m.writes, 0);                    // 1999 ms quiet: not yet
    ps.update(s, 3000);
    EXPECT_EQ(m.writes, 1);                    // 2000 ms quiet: saved
    EXPECT_NEAR(stored_bpm(m, 1), 140.0f, 0.001f);
    EXPECT_FALSE(ps.dirty());
    EXPECT_TRUE(ps.status(3000) == nullptr);   // the guide says auto-save is silent
    ps.update(s, 9000);
    EXPECT_EQ(m.writes, 1);                    // and it does not repeat
}

TEST(AutoSave, FailedWriteIsReportedAndRetriedOnlyLater) {

    EXPECT_EQ(m.writes, 2);                    // retried, failed again

    EXPECT_EQ(m.writes, 3);

}

TEST(HoldToSave, SavesAtOneSecondAndShowsSaved) {

    EXPECT_EQ(m.writes, 1);

TEST(PowerUp, CorruptPresetIsReportedAndLeavesSettingsAlone) {
    MemStorage m; PhaseBoxState s;
    {   PresetSession ps(m); ps.update(s, 0); s.osc.bpm = 133.0f; ps.update(s, 0); ps.update(s, 2000); }
    m.slots[1][10] ^= 0xFF;                     // flip a byte: CRC no longer matches
    PhaseBoxState fresh; fresh.osc.bpm = 120.0f;
    PresetSession ps(m);
    EXPECT_FALSE(ps.load(fresh, 5000));
    EXPECT_TRUE(same(ps.status(5000), "BAD PRESET"));
    EXPECT_NEAR(fresh.osc.bpm, 120.0f, 0.001f);
}

TEST(Slots, SwitchingFlushesPendingEditsLoadsTargetAndKeepsEmptySlotsClean) {
    MemStorage m; PresetSession ps(m); PhaseBoxState s;
    ps.update(s, 0);
    s.osc.bpm = 100.0f; ps.update(s, 0);
    EXPECT_TRUE(ps.select_slot(2, s, 500));      // only 500 ms after the edit
    EXPECT_NEAR(stored_bpm(m, 1), 100.0f, 0.001f);       // edit was not lost: flushed to slot 1
    EXPECT_EQ(ps.slot(), 2);
    EXPECT_TRUE(same(ps.status(500), "EMPTY"));
    EXPECT_NEAR(s.osc.bpm, 100.0f, 0.001f);              // settings carried over into the empty slot
    for (uint32_t t = 516; t < 8000; t += 16) ps.update(s, t);
    EXPECT_EQ(m.slots.count(2), 0u);                     // nothing is written until you edit or hold-save

    s.osc.bpm = 90.0f; ps.update(s, 8000); ps.update(s, 10000);
    EXPECT_NEAR(stored_bpm(m, 2), 90.0f, 0.001f);        // an edit auto-saves into slot 2

    EXPECT_TRUE(ps.select_slot(1, s, 11000));
    EXPECT_NEAR(s.osc.bpm, 100.0f, 0.001f);              // slot 1 comes back as saved
    EXPECT_TRUE(same(ps.status(11000), "LOADED"));
    EXPECT_FALSE(ps.select_slot(0, s, 11000));
    EXPECT_FALSE(ps.select_slot(9, s, 11000));
}

TEST(NoStorageTest, EditsStayUnsavedQuietlyAndHoldSaysNoStorage) {

// ---- reset everything + undo ----
namespace {
PhaseBoxState edited() {
    PhaseBoxState s;
    s.osc.bpm = 150.0f; s.turing_mix = 0.4f; s.macros.destinations[1].midi_channel = 3;
    return s;
}
}

TEST(ResetAll, RestoresDefaultsShowsResetAndAutoSavesThemLikeAnyEdit) {
    MemStorage m; PresetSession ps(m); PhaseBoxState s = edited();
    ps.update(s, 0);                                     // baseline: the edited settings are the saved ones
    ps.reset_to_defaults(s, 100);
    EXPECT_NEAR(s.osc.bpm, 120.0f, 0.001f);
    EXPECT_TRUE(same(ps.status(100), "RESET"));
    ps.update(s, 100);
    EXPECT_TRUE(ps.dirty());
    EXPECT_EQ(m.writes, 0);                              // not written instantly...
    ps.update(s, 2100);
    EXPECT_EQ(m.writes, 1);                              // ...but auto-saved after the usual 2 s
    EXPECT_NEAR(stored_bpm(m, 1), 120.0f, 0.001f);
}

TEST(ResetAll, UndoBringsTheOldSettingsBackEvenAfterTheDefaultsWereSaved) {
    MemStorage m; PresetSession ps(m); PhaseBoxState s = edited();
    ps.update(s, 0);
    ps.reset_to_defaults(s, 100);
    ps.update(s, 100); ps.update(s, 2100);               // defaults are now on disk
    ASSERT_NEAR(stored_bpm(m, 1), 120.0f, 0.001f);
    EXPECT_TRUE(ps.undo_reset(s, 2200));
    EXPECT_NEAR(s.osc.bpm, 150.0f, 0.001f);
    EXPECT_NEAR(s.turing_mix, 0.4f, 0.001f);
    EXPECT_EQ(s.macros.destinations[1].midi_channel, 3);
    EXPECT_TRUE(same(ps.status(2200), "UNDONE"));
    ps.update(s, 2200); ps.update(s, 4300);              // and auto-save writes them back
    EXPECT_NEAR(stored_bpm(m, 1), 150.0f, 0.001f);
}

TEST(ResetAll, UndoIsOnlyOfferedUntilYouChangeSomethingElse) {
    MemStorage m; PresetSession ps(m); PhaseBoxState s = edited();
    ps.update(s, 0);
    ps.reset_to_defaults(s, 100);
    s.osc.bpm = 133.0f;                                  // an edit after the reset
    EXPECT_FALSE(ps.undo_reset(s, 200));
    EXPECT_TRUE(same(ps.status(200), "EDITED SINCE RESET"));
    EXPECT_NEAR(s.osc.bpm, 133.0f, 0.001f);              // the new edit is not thrown away
    EXPECT_FALSE(ps.undo_reset(s, 300));
    EXPECT_TRUE(same(ps.status(300), "NOTHING TO UNDO"));
}

TEST(ResetAll, UndoWithNothingToUndoAndUndoingTwice) {
    MemStorage m; PresetSession ps(m); PhaseBoxState s = edited();
    EXPECT_FALSE(ps.undo_reset(s, 0));
    EXPECT_TRUE(same(ps.status(0), "NOTHING TO UNDO"));
    ps.reset_to_defaults(s, 10);
    EXPECT_TRUE(ps.undo_reset(s, 20));
    EXPECT_FALSE(ps.undo_reset(s, 30));                  // one step only
    EXPECT_NEAR(s.osc.bpm, 150.0f, 0.001f);
}

TEST(ResetAll, ResettingWhenAlreadyAtDefaultsSaysSoAndLeavesNothingToUndo) {
    MemStorage m; PresetSession ps(m); PhaseBoxState s;

TEST(ResetAll, AnAccidentalSecondResetDoesNotLoseTheFirstUndo) {
    MemStorage m; PresetSession ps(m); PhaseBoxState s = edited();

}
